{# `label` and one named submission; with more, a "+N" button opens a menu linking every one. Names the submission whose id is `current_id`, else the first. `submissions`: objects with `.id` and `.name` (SubmissionRef). #} {%- macro submission_links(label, submissions, current_id=None) -%} {%- set named = (submissions | selectattr('id', 'equalto', current_id) | first) if current_id else none -%} {%- set named = named or submissions[0] -%} {{- label }} {{ named.name }} {% if submissions | length > 1 %}
{% for s in submissions %} {{ s.name }} {% endfor %}
{%- endif %} {%- endmacro -%}
